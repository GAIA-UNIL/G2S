# Covariance-model strings and their canonical form.
# Shared by the C++ test (test/geostat/geostat_test.cpp) and the tests of the
# Python and MATLAB model builders, which must produce the same canonical text.
#
# Format: one case per line,
#     <input> ==> <expected>
# where <expected> is either
#     the canonical text that g2s::geostat::ModelSpec::canonical() prints, or
#     !error <fragment>   parsing must fail with a message containing <fragment>.
# The separator is the first " ==> " of the line (it never occurs in a model).
# Spaces around <input> and <expected> are not significant; an empty <input> is
# the empty model. Empty lines and lines starting with '#' are ignored.
# The C++ test also checks that parsing <expected> gives <expected> back.
#
# Canonical rules (details in include/geostat/covarianceModel.hpp and expression.hpp):
#   terms joined by " + "; every term is <sill>*<structure>, sill 1 when absent,
#   parenthesised when it is a sum or difference; lower-case names; "nugget"
#   without parentheses; keyword order range|scale, angle, nu|alpha; vectors
#   "[a, b]", one-element vectors as scalars; constant sub-expressions folded;
#   numbers: integers below 1e15 without decimals, otherwise the shortest
#   "%.{p}g" that reads back to the same double; " + " and " - " spaced,
#   "*" and "/" not; minimal parentheses.

# ---- the examples of the specification
0.1*nugget + 0.6*spherical(range=[40,20], angle=30) + 0.3*exponential(range=100) ==> 0.1*nugget + 0.6*spherical(range=[40, 20], angle=30) + 0.3*exponential(range=100)
nug*nugget + (1-nug)*spherical(range=[L, L/2], angle=theta) ==> nug*nugget + (1 - nug)*spherical(range=[L, L/2], angle=theta)
